'=====================================================================================
' Generate in Web: eksport av skjematisk plan til JSON (planformat v1)
' Limes inn NEDERST i SchemDataModule (bruker modulens private variabler og funksjoner).
' Formatet er beskrevet i docs/planformat.md i repoet dooniem/schemplan.
'=====================================================================================
' Bare prosedyrer her, ingen modulvariabler, så alt kan limes inn nederst i modulen.

Public Sub ExportSchematicJson(dateArr As Variant, SchedNumber As Long, formMultiplier As Double, formTextSize As Long)
    Const SCHEMPLAN_URL As String = "https://dooniem.github.io/schemplan/"
    Const SCHEMPLAN_FORMAT_VERSION As Long = 1
    Dim sb As Collection, laneKey As Variant, taskKey As Variant, lane As cLevel1, taskbox As cLevel2
    Dim taskData As Variant, filePath As String, anonymize As Boolean, answer As VbMsgBoxResult
    Dim filterOutput As Boolean, flag20Only As Boolean, laneShown As Boolean, boxShown As Boolean
    Dim taskIds As Object, firstLane As Boolean, firstBox As Boolean, uid As Variant
    Dim numBoxes As Long, numSkipped As Long, i As Long

    answer = MsgBox("Skal tekstene anonymiseres?" & vbNewLine & vbNewLine & _
        "Ja: lanes, bokser og aktivitetsnavn gjøres om til fyllord med samme lengde." & vbNewLine & _
        "Bruk dette når filen skal deles som testfil." & vbNewLine & vbNewLine & _
        "Nei: ekte tekst, for eget bruk.", vbYesNoCancel + vbQuestion, "Generate in Web")
    If answer = vbCancel Then Exit Sub
    anonymize = (answer = vbYes)

    dateError = False
    generateCalendar dateArr, 2 'Setter pSchemStart og pSchemEnd, tegner ingenting
    If dateError Then Exit Sub
    ParseSchemData False, SchedNumber 'Fyller database og fargelisten
    FagColumn = getFagCol

    filterOutput = SchematicSchedule.chkFilterOutput.value
    flag20Only = SchematicSchedule.chkFlag20.value
    If flag20Only Then filterOutput = False 'Samme regel som i CreateSchematic

    Set sb = New Collection
    Set taskIds = CreateObject("Scripting.Dictionary")

    sb.Add "{" & vbLf
    sb.Add "  ""format"": ""schemplan""," & vbLf
    sb.Add "  ""version"": " & SCHEMPLAN_FORMAT_VERSION & "," & vbLf
    sb.Add "  ""exportedAt"": " & JsonStr(Format(Now, "yyyy-mm-dd\Thh:nn:ss")) & "," & vbLf
    sb.Add "  ""anonymized"": " & JsonBool(anonymize) & "," & vbLf

    ' ---- Plan og innstillinger ----
    sb.Add "  ""plan"": {" & vbLf
    sb.Add "    ""title"": " & JsonStr(IIf(anonymize, "Eksempelplan", CleanSchemHeader(SchematicSchedule.txtSchemHeader.text))) & "," & vbLf
    sb.Add "    ""schemNumber"": " & SchedNumber & "," & vbLf
    sb.Add "    ""calendar"": {" & _
        """mode"": " & JsonStr(IIf(SchemPrintWeeks, "week", "month")) & ", " & _
        """start"": " & JsonDate(pSchemStart) & ", " & _
        """end"": " & JsonDate(pSchemEnd) & ", " & _
        """workingDaysPerWeek"": " & IIf(SchemPrintWeeks, SchemPrintWorkingdays, 5) & ", " & _
        """bottomCalendar"": " & JsonBool(SchematicSchedule.chkExtraCalendar.value) & "}," & vbLf
    sb.Add "    ""layout"": {" & _
        """orientation"": " & JsonStr(IIf(PORTRAIT_MODE, "portrait", "landscape")) & ", " & _
        """heightMode"": " & JsonStr(IIf(PLAN_HEIGHT_MODE = PLAN_HEIGHT_COMPACT, "compact", "fill")) & ", " & _
        """textSize"": " & formTextSize & ", " & _
        """calloutMaxWidthCm"": " & JsonNum(calloutMaxWidth / 28.3465) & ", " & _
        """excelWidthMultiplier"": " & JsonNum(formMultiplier) & "}," & vbLf
    sb.Add "    ""options"": {" & _
        """legend"": " & JsonBool(SchematicSchedule.chkLegend.value) & ", " & _
        """flag20Only"": " & JsonBool(flag20Only) & ", " & _
        """filterOutput"": " & JsonBool(filterOutput) & ", " & _
        """colorBy"": " & JsonStr(IIf(SchemColorBroker = 2, "lane", "trade")) & "}" & vbLf
    sb.Add "  }," & vbLf

    ' ---- Farger per fag: bakgrunn, kant ("#" = ingen), tekst ----
    sb.Add "  ""colors"": [" & vbLf
    For i = 0 To pColorDict.Count - 1
        sb.Add "    " & JsonColorEntry(CStr(pColorDict.keys()(i)), CStr(pColorDict.items()(i))) & _
            IIf(i < pColorDict.Count - 1, ",", "") & vbLf
    Next i
    sb.Add "  ]," & vbLf

    ' ---- Ferier: kalenderunntak over 5 dager som starter i perioden (samme regel som DrawHolidays) ----
    sb.Add "  ""holidays"": [" & JsonHolidays() & "]," & vbLf

    ' ---- Lanes og bokser ----
    sb.Add "  ""lanes"": [" & vbLf
    firstLane = True
    i = 0
    For Each laneKey In database.keys
        i = i + 1
        Set lane = database(laneKey)
        laneShown = True
        If filterOutput Then
            If laneFilterDict.Exists(laneKey) Then If laneFilterDict(laneKey) = False Then laneShown = False
            If laneShown And Not boxFilterDict Is Nothing Then
                If boxFilterDict.Count > 0 Then If Not boxFilterDict.Exists(lane.pOneID) Then laneShown = False
            End If
        End If

        sb.Add IIf(firstLane, "", "," & vbLf) & "    {""id"": " & JsonStr(lane.pOneID) & _
            ", ""name"": " & JsonStr(IIf(anonymize, AnonText(lane.TextName), lane.TextName)) & _
            ", ""fill"": " & JsonColorOrNull(lane.SchemBgColor) & _
            ", ""shown"": " & JsonBool(laneShown) & ", ""boxes"": [" & vbLf
        firstLane = False

        firstBox = True
        For Each taskKey In lane.tasks.keys
            Set taskbox = lane.tasks(taskKey)
            numBoxes = numBoxes + 1
            boxShown = laneShown
            If boxShown And filterOutput Then
                If Not boxFilterDict Is Nothing Then
                    If boxFilterDict.Count > 0 Then If Not boxFilterDict.Exists(taskbox.pTwoID) Then boxShown = False
                End If
            End If
            taskData = getSchemTaskData(taskbox.TaskValue, taskbox.taskExtra, flag20Only)
            If IsEmpty(taskData) Then numSkipped = numSkipped + 1

            For Each uid In taskbox.TaskValue
                If Len(Trim$(CStr(uid))) > 0 Then If Not taskIds.Exists(CStr(uid)) Then taskIds.Add CStr(uid), True
            Next uid

            sb.Add IIf(firstBox, "", "," & vbLf) & "      {""id"": " & JsonStr(taskbox.pTwoID) & _
                ", ""text"": " & JsonStr(IIf(anonymize, AnonText(taskbox.text), taskbox.text)) & _
                ", ""taskIds"": [" & JsonIdList(taskbox.TaskValue) & "]" & _
                ", ""rotation"": " & taskbox.TextRotation & _
                ", ""milestone"": " & JsonMilestone(taskbox) & _
                ", ""shown"": " & JsonBool(boxShown) & _
                ", ""resolved"": " & JsonResolved(taskData) & "}"
            firstBox = False
        Next taskKey
        sb.Add vbLf & "    ]}"
    Next laneKey
    sb.Add vbLf & "  ]," & vbLf

    ' ---- Aktivitetene boksene viser ----
    sb.Add "  ""tasks"": [" & JsonTasks(taskIds, anonymize) & vbLf & "  ]" & vbLf
    sb.Add "}" & vbLf

    filePath = WebExportPath(SchedNumber, anonymize)
    If Not WriteUtf8File(filePath, JoinCollection(sb)) Then
        MsgBox "Klarte ikke å lagre filen:" & vbNewLine & filePath, vbExclamation, "Generate in Web"
        Exit Sub
    End If

    answer = MsgBox("Planen er lagret som:" & vbNewLine & filePath & vbNewLine & vbNewLine & _
        numBoxes & " bokser, " & taskIds.Count & " aktiviteter" & _
        IIf(numSkipped > 0, ", " & numSkipped & " bokser uten gyldige aktiviteter", "") & "." & vbNewLine & vbNewLine & _
        "Åpne web-appen nå? Velg filen over i appen.", vbYesNo + vbInformation, "Generate in Web")
    If answer = vbYes Then CreateObject("Shell.Application").ShellExecute SCHEMPLAN_URL
End Sub

' ---------- Hjelpefunksjoner for eksporten ----------

Private Function WebExportPath(SchedNumber As Long, anonymize As Boolean) As String
    'Ved siden av .mpp-filen. Ligger prosjektet på SharePoint uten synk (adresse, ikke mappe), brukes Nedlastinger.
    Dim folder As String, baseName As String
    folder = SchemProject.Path
    If folder = "" Or LCase$(left$(folder, 4)) = "http" Then folder = Environ$("USERPROFILE") & "\Downloads"
    baseName = SchemProject.Name
    If InStrRev(baseName, ".") > 0 Then baseName = left$(baseName, InStrRev(baseName, ".") - 1)
    If anonymize Then baseName = "Eksempelplan"
    WebExportPath = folder & "\" & baseName & " - skjematisk " & SchedNumber & IIf(anonymize, " (anonymisert)", "") & ".json"
End Function

Private Function WriteUtf8File(filePath As String, content As String) As Boolean
    'UTF-8 uten BOM
    Dim txt As Object, bin As Object
    On Error GoTo fail
    Set txt = CreateObject("ADODB.Stream")
    txt.Type = 2: txt.Charset = "utf-8": txt.Open
    txt.WriteText content
    txt.Position = 3 'Hopp over BOM
    Set bin = CreateObject("ADODB.Stream")
    bin.Type = 1: bin.Open
    txt.CopyTo bin
    bin.SaveToFile filePath, 2 'adSaveCreateOverWrite
    bin.Close: txt.Close
    WriteUtf8File = True
    Exit Function
fail:
    WriteUtf8File = False
End Function

Private Function JoinCollection(c As Collection) As String
    Dim arr() As String, i As Long
    If c.Count = 0 Then Exit Function
    ReDim arr(1 To c.Count)
    For i = 1 To c.Count
        arr(i) = c(i)
    Next i
    JoinCollection = Join(arr, "")
End Function

Private Function JsonStr(ByVal s As String) As String
    Dim i As Long, ch As String, code As Long, out As String
    For i = 1 To Len(s)
        ch = Mid$(s, i, 1)
        code = AscW(ch)
        Select Case code
            Case 34: out = out & "\"""
            Case 92: out = out & "\\"
            Case 10: out = out & "\n"
            Case 13: out = out & "\r"
            Case 9: out = out & "\t"
            Case 0 To 31: out = out & "\u" & Right$("000" & Hex$(code), 4)
            Case Else: out = out & ch
        End Select
    Next i
    JsonStr = """" & out & """"
End Function

Private Function JsonBool(ByVal b As Boolean) As String
    JsonBool = IIf(b, "true", "false")
End Function

Private Function JsonNum(ByVal d As Double) As String
    'Alltid punktum som desimaltegn, uavhengig av regionale innstillinger
    JsonNum = Replace(Format(Round(d, 3), "0.###"), ",", ".")
    If Right$(JsonNum, 1) = "." Then JsonNum = left$(JsonNum, Len(JsonNum) - 1)
End Function

Private Function JsonDate(ByVal d As Variant) As String
    If IsDate(d) Then
        JsonDate = """" & Format(CDate(d), "yyyy-mm-dd") & """"
    Else
        JsonDate = "null"
    End If
End Function

Private Function HexColor(ByVal v As String) As String
    'Excel/VBA-farge (Long, BGR) til "#RRGGBB". Tom streng hvis ugyldig.
    Dim n As Long
    v = Trim$(v)
    If v = "" Or Not IsNumeric(v) Then Exit Function
    n = CLng(v)
    HexColor = "#" & Right$("0" & Hex$(n And &HFF&), 2) & Right$("0" & Hex$((n \ &H100&) And &HFF&), 2) & _
        Right$("0" & Hex$((n \ &H10000) And &HFF&), 2)
End Function

Private Function JsonColorOrNull(ByVal v As String) As String
    Dim h As String
    h = HexColor(v)
    If h = "" Then JsonColorOrNull = "null" Else JsonColorOrNull = """" & h & """"
End Function

Private Function JsonColorEntry(fagName As String, colorStr As String) As String
    Dim parts As Variant, border As String
    parts = Split(colorStr, ",")
    ReDim Preserve parts(0 To 2)
    If Trim$(parts(1)) = "#" Then border = "null" Else border = JsonColorOrNull(CStr(parts(1)))
    JsonColorEntry = "{""name"": " & JsonStr(fagName) & ", ""fill"": " & JsonColorOrNull(CStr(parts(0))) & _
        ", ""border"": " & border & ", ""text"": " & JsonColorOrNull(CStr(parts(2))) & "}"
End Function

Private Function JsonIdList(ids As Variant) As String
    Dim i As Long, out As String, v As String
    If Not IsArray(ids) Then Exit Function
    For i = LBound(ids) To UBound(ids)
        v = Trim$(CStr(ids(i)))
        If IsNumeric(v) And v <> "" Then out = out & IIf(out = "", "", ", ") & CLng(v)
    Next i
    JsonIdList = out
End Function

Private Function JsonMilestone(taskbox As cLevel2) As String
    'taskExtra: 0=planlagt dato(1)/frist(0), 1=flagg(0)/diamant(1), 2=loddrett linje, 3=farge 1-6, 4=rotert tekst, 5=vis dato
    Dim e As Variant, colorNames As Variant, c As Long
    If Not taskbox.HasTaskExtra Then
        JsonMilestone = "null"
        Exit Function
    End If
    e = taskbox.taskExtra
    If UBound(e) < 5 Then ReDim Preserve e(0 To 5)
    colorNames = Array("", "red", "yellow", "green", "blue", "orange", "black")
    c = CLng(val(e(3)))
    If c < 1 Or c > 6 Then c = 4
    JsonMilestone = "{""dateSource"": " & JsonStr(IIf(val(e(0)) = 0, "deadline", "scheduled")) & _
        ", ""symbol"": " & JsonStr(IIf(val(e(1)) = 1, "diamond", "flag")) & _
        ", ""verticalLine"": " & JsonBool(val(e(2)) = 1) & _
        ", ""color"": " & JsonStr(CStr(colorNames(c))) & _
        ", ""rotateText"": " & JsonBool(val(e(4)) = 1) & _
        ", ""showDate"": " & JsonBool(val(e(5)) = 1) & "}"
End Function

Private Function JsonResolved(taskData As Variant) As String
    'Det Excel ville tegnet: Array(start, finish, milepæl, fag). Empty = boksen tegnes ikke (Flag20 eller ingen aktiviteter)
    If IsEmpty(taskData) Then
        JsonResolved = "null"
    Else
        JsonResolved = "{""start"": " & JsonDate(taskData(0)) & ", ""finish"": " & JsonDate(taskData(1)) & _
            ", ""isMilestone"": " & JsonBool(CBool(taskData(2))) & ", ""trade"": " & JsonStr(CStr(taskData(3))) & "}"
    End If
End Function

Private Function JsonHolidays() As String
    Dim item As Variant, out As String
    On Error Resume Next
    For Each item In SchemProject.Calendar.Exceptions
        If DateDiff("d", item.Start, item.Finish) > 5 Then
            If item.Start >= pSchemStart And item.Start <= pSchemEnd Then
                out = out & IIf(out = "", "", ", ") & "{""start"": " & JsonDate(DateValue(item.Start)) & _
                    ", ""end"": " & JsonDate(DateValue(item.Finish)) & "}"
            End If
        End If
    Next item
    JsonHolidays = out
End Function

Private Function JsonTasks(taskIds As Object, anonymize As Boolean) As String
    Dim key As Variant, t As task, out As String, nm As String, trade As String, dl As Variant
    For Each key In taskIds.keys
        Set t = Nothing
        On Error Resume Next
        Set t = SchemProject.tasks.UniqueID(CLng(key))
        On Error GoTo 0
        If Not t Is Nothing Then
            nm = t.Name
            If anonymize Then nm = AnonText(nm)
            trade = ""
            If FagColumn > 0 Then
                On Error Resume Next
                trade = t.GetField(FagColumn)
                On Error GoTo 0
            End If
            dl = t.Deadline
            out = out & IIf(out = "", "", ",") & vbLf & "    {""uid"": " & t.UniqueID & _
                ", ""name"": " & JsonStr(nm) & _
                ", ""start"": " & JsonDate(DateValue(t.Start)) & _
                ", ""finish"": " & JsonDate(DateValue(t.Finish)) & _
                ", ""deadline"": " & IIf(IsDate(dl), JsonDate(DateValue(dl)), "null") & _
                ", ""milestone"": " & JsonBool(t.Milestone) & _
                ", ""summary"": " & JsonBool(t.Summary) & _
                ", ""flag20"": " & JsonBool(t.Flag20) & _
                ", ""percentComplete"": " & CLng(val(t.PercentComplete)) & _
                ", ""trade"": " & IIf(trade = "", "null", JsonStr(trade)) & "}"
        End If
    Next key
    JsonTasks = out
End Function

Private Function AnonText(ByVal s As String) As String
    'Bytter hver bokstav med fyllord med samme lengde og store/små bokstaver.
    'Tall, mellomrom og tegn som | / * beholdes, slik at linjeskift, callout og bredder blir like.
    Const FILLER As String = "loremipsumdolorsitametconsecteturadipiscingelitseddoeiusmodtemporincididuntutlaboreetdoloremagnaaliqua"
    Dim i As Long, k As Long, ch As String, rep As String, out As String
    For i = 1 To Len(s)
        ch = Mid$(s, i, 1)
        If LCase$(ch) <> UCase$(ch) Then
            rep = Mid$(FILLER, (k Mod Len(FILLER)) + 1, 1)
            If ch = UCase$(ch) Then rep = UCase$(rep)
            out = out & rep
            k = k + 1
        Else
            out = out & ch
        End If
    Next i
    AnonText = out
End Function
